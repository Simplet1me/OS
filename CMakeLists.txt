cmake_minimum_required(VERSION 3.20)
project(OSExperiments LANGUAGES CXX)

add_library(banker src/banker.cpp)
target_include_directories(banker PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(banker PUBLIC cxx_std_17)
set_target_properties(banker PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(banker PUBLIC /W4 /permissive- /utf-8)
else()
    target_compile_options(banker PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(exp_deadlock exp/exp_deadlock.cpp)
target_link_libraries(exp_deadlock PRIVATE banker)

include(CTest)
if(BUILD_TESTING)
    add_executable(test_banker test/test_banker.cpp)
    target_link_libraries(test_banker PRIVATE banker)
    add_test(NAME banker COMMAND test_banker)
endif()
